 let stage=$state(),frame=$state(),gpuCanvas=$state(),available=$state({w:800,h:500}),imageSize=$state({w:1600,h:1200});let fit=$derived(Math.min(available.w*.96/imageSize.w,available.h*.96/imageSize.h));

  return()=>{live=false;controller.abort();clearTimeout(sourceTimer);clearTimeout(preciseTimer);clearTimeout(busyTimer);cancelAnimationFrame(drawFrame);};
 });

 function key(e){if(e.key==='Enter'&&studio.tool==='crop')applyCrop();if(e.key==='Escape')selection=null;}
</script>
